cmake_minimum_required(VERSION 3.24)
project(lumen VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LUMEN_CUDA "Build the CUDA wavefront backend" OFF)
option(LUMEN_VIEWER "Build the GLFW OpenGL viewer" OFF)
find_package(Threads REQUIRED)
add_library(lumen_core src/scene.cpp src/render.cpp src/image.cpp)
target_include_directories(lumen_core PUBLIC include)
target_link_libraries(lumen_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(lumen_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4>)
else()
  target_compile_options(lumen_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
endif()
if(LUMEN_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_sources(lumen_core PRIVATE src/wavefront.cu)
  set_target_properties(lumen_core PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_definitions(lumen_core PUBLIC LUMEN_HAS_CUDA)
  target_link_libraries(lumen_core PUBLIC CUDA::cudart)
endif()
add_executable(lumen src/main.cpp)
target_link_libraries(lumen PRIVATE lumen_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(lumen_tests tests/core_test.cpp)
  target_link_libraries(lumen_tests PRIVATE lumen_core)
  add_test(NAME core COMMAND lumen_tests)
endif()
if(LUMEN_VIEWER)
  find_package(OpenGL REQUIRED)
  find_package(glfw3 3.3 REQUIRED)
  add_executable(lumen_viewer src/viewer.cpp)
  target_link_libraries(lumen_viewer PRIVATE lumen_core glfw OpenGL::GL)
endif()
